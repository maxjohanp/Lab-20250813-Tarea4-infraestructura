---- BLOQUE 1: nombre e interfaces (port1 = WAN por DHCP, port2 = LAN) ----
config system global
    set hostname FW-0813
end
config system interface
    edit port1
        set alias WAN
        set role wan
        set mode dhcp
    next
    edit port2
        set alias LAN
        set role lan
        set mode static
        set ip 10.8.13.1 255.255.255.252
        set allowaccess ping http https ssh
    next
end

---- BLOQUE 2: ruta hacia la LAN (pasa por el router) ----
config router static
    edit 1
        set dst 10.20.25.0 255.255.255.0
        set gateway 10.8.13.2
        set device port2
    next
end

---- BLOQUE 3: politica de salida a Internet con NAT ----
config firewall policy
    edit 1
        set name "LAN-a-Internet"
        set srcintf port2
        set dstintf port1
        set srcaddr all
        set dstaddr all
        set action accept
        set schedule always
        set service ALL
        set nat enable
    next
end

---- BLOQUE 4: comprobar ----
get system status
get system interface physical
get router info routing-table all
execute ping 10.8.13.2
execute ping 8.8.8.8
